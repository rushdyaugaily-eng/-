# 1. Initialize your project folder as a Git repository
git init

# 2. Add all your project files to staging
git add .

# 3. Commit the files to your local repository history
git commit -m "Initial commit"
# 1. Link your local project to your GitHub remote repository
git remote add origin https://github.com

# 2. Rename your default branch to 'main'
git branch -M main

# 3. Push your local files up to GitHub
git push -u origin main
